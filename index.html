    async function sendBatchToGas(batchData, tableNo, userName, targetFolderId, userMessage) {

      const payload = { chunk: batchData, tableNo: tableNo, userName: userName, targetFolderId: targetFolderId, userMessage: userMessage };

      const response = await fetch(GAS_WEB_APP_URL, {

        method: "POST", redirect: "follow",

        headers: { "Content-Type": "text/plain;charset=utf-8" },

        body: JSON.stringify(payload)

      });

      if (!response.ok) throw new Error(`通信エラー (${response.status})`);

      return await response.json();

    }


    async function startUpload() {

        return;

      document.getElementById('status').innerText = '';


      const totalPhotos = selectedFiles.length; let compressedCount = 0; let uploadedPhotoCount = 0; let targetFolderId = null;


      try {

        updateProgress(0, `最適化中 (0/${totalPhotos}枚)...`);

        const fileDataArray = await Promise.all(selectedFiles.map(file => 

          new Promise((resolve, reject) => {

            const reader = new FileReader();

            reader.onload = (e) => {

              const img = new Image();

              img.onload = () => {

                let w = img.width, h = img.height;

                if (w > h) { if (w > 2000) { h = Math.round(h * 2000 / w); w = 2000; } } else { if (h > 2000) { w = Math.round(w * 2000 / h); h = 2000; } }

                const cvs = document.createElement('canvas'); cvs.width = w; cvs.height = h;

                cvs.getContext('2d').drawImage(img, 0, 0, w, h);

                compressedCount++;

                updateProgress((compressedCount / totalPhotos) * 30, `最適化中 (${compressedCount}/${totalPhotos}枚)...`);

                resolve({ base64Data: cvs.toDataURL('image/jpeg', 0.85), fileName: file.name });

              };

              img.onerror = reject; img.src = e.target.result;

            };

            reader.onerror = reject; reader.readAsDataURL(file);

          })

        ));


        saveUploadHistory(tableNo, userName, fileDataArray);

        const CHUNK_SIZE = 3;


        for (let i = 0; i < fileDataArray.length; i += CHUNK_SIZE) {

          const chunk = fileDataArray.slice(i, i + CHUNK_SIZE);

          if (i + CHUNK_SIZE >= fileDataArray.length) {

            updateProgress(100, `送信完了 (${totalPhotos}/${totalPhotos}枚)`);

            setTimeout(() => navigate('thankyou'), 300);

      }
