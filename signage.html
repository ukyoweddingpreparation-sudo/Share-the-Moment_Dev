<body>


    <div class="slide-container" id="slideContainer"></div>

    <div class="status-badge" id="statusBadge">システム接続中...</div>


    <script>

        // ⚙️ 設定：ご自身の環境に合わせて書き換えてください

        const GAS_WEB_APP_URL = "YOUR_GAS_WEB_APP_URL"; // デプロイしたGASのWebアプリURL

        const FIREBASE_DB_URL = "https://signageresource-syncsystem-default-rtdb.firebaseio.com/trigger.json";


        let images = [];

        let currentIndex = 0;

        let imgElement1, imgElement2;

        let useFirstImage = true;

        let slideTimer = null;

        let lastKnownTimestamp = null;


        // 初期化処理

        async function initViewer() {

            const container = document.getElementById('slideContainer');

            

            imgElement1 = document.createElement('img');

            imgElement2 = document.createElement('img');

            container.appendChild(imgElement1);

            container.appendChild(imgElement2);


            // 初回データ取得＆スタート

            await fetchImagesAndRefresh(true);


            // 4秒ごとに次のスライドへ切り替え

            slideTimer = setInterval(nextSlide, 4000);

            

            // 3秒ごとにFirebaseのトリガーを軽量チェック

            setInterval(checkFirebaseTrigger, 3000);

        }


        // Firebaseのタイムスタンプを確認し、更新があれば画像リストを再取得

        async function checkFirebaseTrigger() {

            try {

                const response = await fetch(FIREBASE_DB_URL);

                const data = await response.json();

                

                if (data && data.timestamp) {

                    if (lastKnownTimestamp === null) {

                        lastKnownTimestamp = data.timestamp;

                    } else if (data.timestamp > lastKnownTimestamp) {

                        console.log('新着写真の合図を検知しました！即時同期します。');

                        lastKnownTimestamp = data.timestamp;

                        

                        await fetchImagesAndRefresh(false, true);


                        // 4秒タイマーをリセットして仕切り直し

                        clearInterval(slideTimer);

                        slideTimer = setInterval(nextSlide, 4000);

                    }

                }

            } catch (e) {

                console.error('Firebaseの確認に失敗しました', e);

            }

        }


        // GASから最新の画像一覧を取得して画面を更新

        async function fetchImagesAndRefresh(isInitial = false, isInterrupt = false) {

            try {

                const response = await fetch(GAS_WEB_APP_URL);

                const data = await response.json();

                

                if (data.status === 'success' && data.images.length > 0) {

                    images = data.images;


                    if (isInitial) {

                        showImage(currentIndex);

                        document.getElementById('statusBadge').innerText = `写真総数: ${images.length}枚`;

                    } else if (isInterrupt) {

                        currentIndex = 0;

                        showImage(currentIndex);

                        document.getElementById('statusBadge').innerText = `✨新着写真を受信！ (総数: ${images.length}枚)`;

                    }

                } else if (data.status === 'success' && data.images.length === 0) {

                    document.getElementById('statusBadge').innerText = 'フォルダに写真がありません';

                }

            } catch (e) {

                console.error('画像データの取得に失敗しました', e);

                document.getElementById('statusBadge').innerText = '通信エラー（再試行中...）';

            }

        }


        // 指定インデックスの画像を表示

        function showImage(index) {

            if (images.length === 0) return;


            const imageUrl = images[index].url;


            if (useFirstImage) {

                imgElement1.src = imageUrl;

                imgElement1.classList.add('active');

                imgElement2.classList.remove('active');

            } else {

                imgElement2.src = imageUrl;

                imgElement2.classList.add('active');

                imgElement1.classList.remove('active');

            }


            useFirstImage = !useFirstImage;

        }


        // 通常の次のスライドへ

        function nextSlide() {

            if (images.length === 0) return;

            currentIndex = (currentIndex + 1) % images.length;

            showImage(currentIndex);

        }


        window.onload = initViewer;

    </script>
